extends SceneTree
const Core = preload("res://addons/estragon/runtime/node/internal/layout_core.gd")
const Length = preload("res://addons/estragon/runtime/node/internal/length.gd")
func _initialize():
	var parent = Core.new()
	parent.configure(Vector3(600,400,0),0,false,1)
	var proxy = Core.new()
	proxy.configure(Vector3(-1,-1,0),0,false,0,Vector3.ZERO,Vector3.ZERO,Vector4.ZERO,Vector4.ZERO,true)
	var panel = Core.new()
	panel.configure(Vector3(-1,-1,0),0,false,1)
	panel.set_lengths({"width":Length.parse(600,"extent"),"height":Length.parse("auto","extent")},1,0,1,1)
	var leaf = Core.new()
	leaf.set_lengths({"width":Length.parse("50%","extent"),"height":Length.parse("50%","extent")},0,0,1,0)
	panel.set_children([leaf])
	proxy.set_children([panel])
	parent.set_children([proxy])
	parent.prepare(Vector3(-1,-1,-1),Vector2.ZERO)
	parent.arrange(parent.measure())
	print("DEEP_FLEX ",panel.size," ",leaf.size)
	quit(0 if leaf.size==Vector3(300,200,0) else 1)
