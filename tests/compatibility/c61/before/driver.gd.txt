extends SceneTree
const Compiler = preload("res://addons/estragon/compiler/compiler.gd")
var _failed := false
var _checks := 0
func _initialize() -> void:
	_run.call_deferred()
func _check(value: bool, message: String) -> void:
	_checks += 1
	print("C61_CHECK ", message, ": ", value)
	if not value:
		_failed = true
		push_error("Helper calls: " + message)
func _run() -> void:
	var source := FileAccess.get_file_as_string("res://tests/fixtures/helper-calls.gogo")
	var compiled := Compiler.compile(source, "tests/fixtures/helper-calls.gogo")
	_check(compiled.ok, "raw-free Gogo compiles")
	if not compiled.ok:
		quit(1)
		return
	var script := GDScript.new()
	script.resource_path = "res://HelperCalls.gd"
	script.source_code = compiled.code
	_check(script.reload() == OK, "generated native code reloads")
	var instance = script.new()
	var first = instance.instance_lambda()
	_check(first is EstragonRect and first.width == 7.0, "instance lambda returns actual component")
	if first == null:
		quit(1)
		return
	_check(instance.initial.width == 2.0 and instance.initial.height == 3.0, "member initializer")
	_check(instance.initializer_callback.call().width == 4.0, "initializer lambda")
	_check(script.static_method().width == 9.0, "static method")
	_check(script.static_lambda().width == 8.0, "static lambda")
	var nested = instance.nested()
	_check(nested.gap == 1.0 and nested.get_children()[0].width == 2.0 and nested.get_children()[1].width == 3.0, "nested lambda expressions execute")
	_check(instance.events == [1, 2, 3], "authored eager evaluation order")
	var button = instance.button()
	var callbacks: Array = button.pressed.get_connections()
	_check(callbacks.size() == 1 and callbacks[0].callable == Callable(instance, "on_pressed"), "original callback identity")
	button.pressed.emit()
	_check(instance.events == [1, 2, 3, 4], "lambda-built button invokes handler")
	var owner := EstragonContainer.new()
	owner.add_child(button)
	owner.remove_child(button)
	_check(button.is_disposed() and button.pressed.get_connections().is_empty(), "normal ownership cleans callback")
	var repeat := Compiler.compile(source, "tests/fixtures/helper-calls.gogo")
	_check(repeat.code == compiled.code and repeat.source_map.serialize() == compiled.source_map.serialize(), "deterministic generated bytes/maps")
	print("C61_ASSERTIONS ", _checks)
	if not _failed: print("C61_HELPER_CALLS_OK")
	quit(1 if _failed else 0)
